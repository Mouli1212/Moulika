from PIL import Image, ImageDraw, ImageFont

draw.rectangle((50, 130, 950, 380), outline="black", width=5)
draw.text((80, 160), "PANEL 1", fill="black")

          "Luffy sails across the ocean searching for treasure!",
          fill="black")

# Panel 2
draw.rectangle((50, 410, 950, 660), outline="black", width=5)

          "Luffy discovers a mysterious island.",
